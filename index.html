const express = require("express"); const fs = require("fs"); const path = require("path"); const app = express(); const PORT = process.env.PORT || 3000; app.use(express.json()); const DATA_FILE = path.join(__dirname, "anime-data.json"); /* ========================= DATABASE ========================= */ function loadData() { if (!fs.existsSync(DATA_FILE)) { const initialData = { anime: [ { id: "naruto", title: "Naruto", description: "Anime description", poster: "", episodes: [ { id: "naruto-episode-1", number: 1, title: "Episode 1", streamLink: "", quality: "1080p" } ] } ] }; fs.writeFileSync( DATA_FILE, JSON.stringify(initialData, null, 2) ); } return JSON.parse(fs.readFileSync(DATA_FILE, "utf8")); } function saveData(data) { fs.writeFileSync( DATA_FILE, JSON.stringify(data, null, 2) ); } /* ========================= SEARCH ANIME ========================= */ app.get("/api/search", (req, res) => { const query = String(req.query.q || "") .trim() .toLowerCase(); const data = loadData(); if (!query) { return res.json({ success: true, results: data.anime }); } const results = data.anime.filter(anime => anime.title.toLowerCase().includes(query) ); res.json({ success: true, results }); }); /* ========================= GET ANIME ========================= */ app.get("/api/anime/:id", (req, res) => { const data = loadData(); const anime = data.anime.find( item => item.id === req.params.id ); if (!anime) { return res.status(404).json({ success: false, error: "Anime not found" }); } res.json({ success: true, anime }); }); /* ========================= GET EPISODE ========================= */ app.get("/api/episode/:id", (req, res) => { const data = loadData(); for (const anime of data.anime) { const episode = anime.episodes.find( ep => ep.id === req.params.id ); if (episode) { return res.json({ success: true, anime: { id: anime.id, title: anime.title, poster: anime.poster }, episode }); } } res.status(404).json({ success: false, error: "Episode not found" }); }); /* ========================= ADD ANIME ========================= */ app.post("/api/anime", (req, res) => { const { id, title, description, poster } = req.body; if (!id || !title) { return res.status(400).json({ success: false, error: "id and title are required" }); } const data = loadData(); if (data.anime.some(a => a.id === id)) { return res.status(400).json({ success: false, error: "Anime already exists" }); } const anime = { id, title, description: description || "", poster: poster || "", episodes: [] }; data.anime.push(anime); saveData(data); res.json({ success: true, anime }); }); /* ========================= ADD EPISODE ========================= */ app.post("/api/anime/:id/episode", (req, res) => { const { number, title, streamLink, quality } = req.body; const data = loadData(); const anime = data.anime.find( a => a.id === req.params.id ); if (!anime) { return res.status(404).json({ success: false, error: "Anime not found" }); } if (!number || !title) { return res.status(400).json({ success: false, error: "Episode number and title are required" }); } const episodeId = `${anime.id}-episode-${number}`; const episode = { id: episodeId, number: Number(number), title, streamLink: streamLink || "", quality: quality || "1080p" }; anime.episodes.push(episode); anime.episodes.sort( (a, b) => a.number - b.number ); saveData(data); res.json({ success: true, episode }); }); /* ========================= STREAM ENDPOINT ========================= */ app.get("/api/stream/:episodeId", (req, res) => { const data = loadData(); for (const anime of data.anime) { const index = anime.episodes.findIndex( ep => ep.id === req.params.episodeId ); if (index !== -1) { const episode = anime.episodes[index]; const sources = []; if (episode.streamLink) { sources.push({ server: "Primary", quality: episode.quality, url: episode.streamLink }); } return res.json({ success: true, anime: { id: anime.id, title: anime.title, poster: anime.poster }, episode, sources, previous: anime.episodes[index - 1]?.id || null, next: anime.episodes[index + 1]?.id || null }); } } res.status(404).json({ success: false, error: "Stream not found" }); }); /* ========================= FRONTEND ========================= */ app.get("/", (req, res) => { res.send(` My Anime Website

`); }); /* ========================= START SERVER ========================= */ app.listen(PORT, () => { console.log( \`Server running on http://localhost:\${PORT}\` ); });
